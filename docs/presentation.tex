%   3. schéma de l'archi retenue .... diapositive 16

%   5. schéma de l'archi cible ...... diapositive 20

\begin{frame}{Deux accès à Blob Storage, pas un}
  \framesubtitle{Pourquoi le binding ne porte que la fraîcheur}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{carte}{signal}{Binding · 23 Ko}
        Relu \textbf{à chaque appel} : exactement ce qu'il faut pour la fenêtre de fraîcheur, qui change toutes les heures.
      \end{carte}

    \begin{column}{0.48\textwidth}

      \end{carte}
    \end{column}

\begin{frame}{Contexte}
  \framesubtitle{Ce qui est demandé, avec quelles données}
  \begin{itemize}
    \item Proposer \pointfort{5 articles} à chaque lecteur, pour l'inciter à lire
          davantage.
    \item Pas de données propriétaires : jeu public Globo.com —
          \textbf{2 988 181} clics, \textbf{322 897} lecteurs,
          \textbf{364 047} articles.
    \item[] \vspace{0.3em}
    \item \pointfort{Contrainte majeure : le jeu est anonymisé.}
      \begin{itemize}
        \item aucun titre, aucun texte : un article n'est qu'un vecteur de
              250 dimensions et un identifiant ;
        \item aucune note, aucun avis : seulement des clics, donc un signal
              implicite ;
        \item conséquence — tout se fait sans lire le contenu, et l'interface ne
              peut afficher que des identifiants.
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{L'application en quatre gestes}
  \framesubtitle{Description fonctionnelle — ce que le lecteur peut faire}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \scriptsize
      \begin{enumerate}\itemsep4pt
        \item \textbf{Recevoir 5 articles.} Le lecteur est choisi, le service
              renvoie cinq identifiants — avec leur note en étoiles, leur nombre
              de lecteurs, leur catégorie et leur date.
        \item \textbf{Marquer un article comme lu.} La lecture est enregistrée,
              le profil s'enrichit et la liste suivante change —
              \pointfort{sans ré-entraînement}.
        \item \textbf{Parcourir le catalogue.} 364 047 articles, quatre tris,
              filtre par note, articles déjà lus masqués.
        \item \textbf{S'inscrire.} Un nouveau lecteur reçoit un identifiant, une
              région et un profil de départ, et il est recommandé
              \pointfort{immédiatement}.
      \end{enumerate}
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering
      \includegraphics[width=\linewidth]{figures/app_recommandations.png}
    \end{column}
  \end{columns}
  \aretenir{Une seule fonction est demandée — \pointfort{cinq articles par
    lecteur} — mais elle doit répondre aussi pour un lecteur inscrit il y a une
    seconde, dont le service ne sait rien.}
\end{frame}

\begin{frame}{Approche 2 — Collaboratif ALS (\texttt{implicit})}
  \framesubtitle{Factorisation pour feedback implicite}
  \scriptsize
  Toutes les cases vides sont traitées comme des \textbf{négatifs pondérés} :
  une formulation adaptée à des clics, pas à des notes.

  \vspace{0.8em}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \textcolor{vert}{\textbf{Avantages}}
      \begin{itemize}\itemsep2pt
        \item capte des goûts \textbf{latents} qu'aucune similarité de contenu ne
              voit ;
        \item effet de surprise : recommande hors du thème déjà lu ;
        \item personnalisation \textbf{95 \%}.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textcolor{red!70!black}{\textbf{Inconvénients}}
      \begin{itemize}\itemsep2pt
        \item HitRate@5 sur test = \textbf{0,0415} ;
        \item \textbf{cold start} : inutile pour un lecteur ou un article nouveau ;
        \item limite structurelle — un article publié il y a deux heures n'a
              presque aucune co-lecture, le collaboratif ne peut pas le proposer ;
        \item réglages sensibles et \pointfort{couplés} entre eux.
      \end{itemize}
    \end{column}
  \end{columns}
  \aretenir{Balayés \textbf{séparément} sur validation, les deux gagnants étaient
    72 h et 16 facteurs. Croisés, ils donnent \textbf{0,0110} — la pire des quatre
    combinaisons ; 24 h + 16 facteurs donne \textbf{0,0345}. On ne juxtapose pas
    des optima partiels.}
\end{frame}

\begin{frame}{Approche 3 — Collaboratif SVD (\texttt{Surprise})}
  \framesubtitle{Trois définitions de note, trois résultats}
  \centering
  \scriptsize
  \begin{tabular}{@{}lccl@{}}
    \toprule
    \textbf{Définition de la note} & \textbf{HitRate@5} & \textbf{Person.} &
    \textbf{Verdict} \\
    \midrule
    Étoiles de l'article (log des clics) & 0,0015 & 23 \% &
      ne classe pas \\
    Binaire + 1 négatif échantillonné   & 0,0080 & 46 \% & insuffisant \\
    Binaire + 4 négatifs échantillonnés & 0,0815\,$^*$ & 34 \% &
      meilleure variante \\
    \bottomrule
  \end{tabular}

  \vspace{0.8em}
  \raggedright\scriptsize
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \textcolor{vert}{\textbf{Avantages}}
      \begin{itemize}\itemsep2pt
        \item bien posée — binaire + 4 négatifs — c'est la meilleure des trois
              variantes sur validation ;
        \item implémentation directe, entraînement rapide, peu de réglages.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textcolor{red!70!black}{\textbf{Inconvénients}}
      \begin{itemize}\itemsep2pt
        \item \pointfort{ne se reproduit pas} : 0,0815 en validation, mais
              \textbf{0,0220} sur test — un facteur 3,7 ;
        \item les étoiles ne classent pas du tout (0,0015) : la note ne dépend que
              de l'article, le modèle apprend « quels articles sont lus », pas
              « qui lit quoi » ;
        \item même cold start que l'ALS ; personnalisation 34 \% seulement.
      \end{itemize}
    \end{column}
  \end{columns}
  \aretenir{Même bibliothèque, même algorithme : c'est la \pointfort{formulation
    du problème} qui change tout, pas l'outil. Et c'est une période de test
    intacte qui révèle qu'un réglage ne s'est pas reproduit.}
\end{frame}

\begin{frame}{Comment nous avons su que les premiers chiffres étaient faux}
  \framesubtitle{Découpage temporel 60 / 20 / 20}
  \small
  \begin{itemize}
    \item Première évaluation : ALS entraîné sur 100 \% des clics, puis mesuré en
          retirant le dernier clic de chaque lecteur — un clic que le modèle avait
          déjà vu. HitRate@5 = \textbf{0,2415}.
    \item Avec un découpage temporel correct, il tombe à \textbf{0,0415} —
          \pointfort{un facteur 5,8} de précision imaginaire, invisible sans
          découpage.
    \item La popularité, elle, ne profite pas de la fuite : son classement ne
          dépend pas du lecteur. Seuls les modèles \emph{personnalisés} peuvent
          recopier ce qu'ils ont appris d'un lecteur.
    \item Temporel et non aléatoire : sur un flux d'actualité, un découpage
          aléatoire laisse le modèle apprendre des clics \emph{postérieurs} à ceux
          qu'il doit prédire.
    \item Le réglage se fait sur la validation ; le test ne sert qu'à la mesure
          finale, une seule fois.
  \end{itemize}

  \vspace{0.5em}
  \centering
  \begin{tikzpicture}[every node/.style={font=\scriptsize, text=white,
        minimum height=0.9cm, align=center}]
    \node[fill=accent,  minimum width=6cm]   (a) {60 \% entraînement};
    \node[fill=accent2, minimum width=2.6cm, right=0pt of a] (b) {20 \% validation};
    \node[fill=primary, minimum width=2.6cm, right=0pt of b] (c) {20 \% test};
  \end{tikzpicture}

  {\scriptsize\color{gris} le temps s'écoule de gauche à droite — aucune fuite du
   futur vers le passé}
\end{frame}

  \vspace{0.3em}

\begin{frame}{Quatre critères, pas un seul}
  \framesubtitle{La précision seule désigne toujours la popularité}
  \centering
  \scriptsize
  \begin{tabular}{@{}llrr@{}}
    \toprule
    \textbf{Critère} & \textbf{Ce qu'il mesure} & \textbf{Popularité} &
    \textbf{Contenu} \\
    \midrule
    HitRate@5        & un article recommandé a été lu      & 0,2525 & 0,0200 \\
    Couverture       & part du catalogue exposée           & 0,003 \% & 0,27 \% \\
    Personnalisation & différence entre deux listes        & 4 \%   & 96 \% \\
    Recall@5         & part du futur devinée (plafond 0,76)& 0,0555 & 0,0034 \\
    \bottomrule
  \end{tabular}

  \vspace{0.8em}
  \raggedright\small
  \begin{itemize}\itemsep3pt
    \item La popularité est 12 fois plus précise, mais expose \textbf{11 articles
          sur 364 047} : un éditeur y perd son catalogue.
    \item Aucun critère ne sauve le \pointfort{SVD sur étoiles} : HitRate nul et
          personnalisation 12 \%, contre 96 \% pour le contenu. Il ne trouve rien
          \emph{et} ne distingue presque pas les lecteurs — parce que sa note ne
          dépend que de l'article.
  \end{itemize}
\end{frame}

\begin{frame}{Résultat final sur la période de test}
  \framesubtitle{Chaque approche dans sa meilleure configuration}
  \centering
  \includegraphics[width=0.97\textwidth]{figures/compromis.pdf}

  \vspace{0.3em}
  \raggedright\footnotesize
  \begin{itemize}\itemsep2pt
    \item Retenu : \pointfort{4 populaires (1 h) + 1 contenu} — la place donnée au
          contenu ne coûte rien en précision (écart non significatif) et
          \textbf{multiplie la couverture par 38}.
    \item Les deux collaboratifs restent en bas à gauche : un ordre de grandeur

          le contenu. Disponibles dans l'application, ils ne sont pas servis.
  \end{itemize}
\end{frame}

\begin{frame}{Un lecteur dont on ne sait rien}
  \framesubtitle{Le cas le plus fréquent sur un portail d'actualité}

    \node[bloc, fill=accent,  right=of a] (b) {Articles récents\\les plus lus};
    \node[bloc, fill=accent2, right=of b] (c) {Région connue :\\articles récents\\de sa région};
    \draw[-{Latex[length=2mm]}, gris, thick] (a) -- (b);
    \draw[-{Latex[length=2mm]}, gris, thick] (b) -- (c);
  \end{tikzpicture}

  \vspace{1em}
  \raggedright\small
  \begin{itemize}\itemsep4pt
    \item \textbf{Cascade de repli} : région croisée avec la fraîcheur, puis
          fraîcheur, puis région, puis historique complet. Aucune requête ne
          renvoie de liste vide.
    \item La région est la seule information exploitable sur un lecteur inconnu :
          27 régions retenues, celles ayant assez de clics pour un classement.
    \item Dès le premier article lu, le profil de contenu prend le relais —
          \pointfort{sans aucun ré-entraînement}.
  \end{itemize}
\end{frame}

    \item La projection ACP est sérialisée avec le catalogue (\textbf{51 Ko}) :
          sans elle, intégrer un article imposerait de réajuster l'ACP, donc de
          recalculer les 364 047 vecteurs existants.
    \item Vérifié : trois articles ajoutés reçoivent les identifiants
          364047--364049 et \textbf{entrent immédiatement} dans le top-5 d'un
          lecteur proche.
    \item L'article n'a ni étoiles ni facteurs collaboratifs — il n'a aucun clic.
          Il est recommandé par le contenu, seul signal disponible à cet instant.
  \end{itemize}
\end{frame}

\begin{frame}{Partie 3 — Architecture retenue : l'architecture 2}
  \framesubtitle{Serverless \emph{sans} API dédiée — la Function lit Blob Storage elle-même}
  \centering
  \begin{tikzpicture}[every node/.style={font=\scriptsize, align=center},
      app/.style={draw=gris, thick, rounded corners, text width=2.4cm,
                  minimum height=1.2cm, fill=clair, text=primary},
      fn/.style={draw=primary, very thick, rounded corners, text width=3.9cm,
                 minimum height=1.6cm, fill=primary, text=white},
      hot/.style={draw=accent2, very thick, rounded corners, text width=3.7cm,
                  minimum height=1.15cm, fill=accent2, text=white},
      cold/.style={draw=accent, thick, rounded corners, text width=3.7cm,
                   minimum height=1.15cm, fill=accent, text=white}]

    \node[fn] (fn) {\textbf{Azure Function}\\\texttt{/api/recommend}\\
                    Recommender — \textbf{numpy seul}};
    \node[app, left=12mm of fn] (app) {Application\\locale};
    \node[hot,  right=14mm of fn, yshift=11mm]  (hot)
      {\textbf{Fraîcheur} — 23 Ko\\\textbf{blob input binding}\\relu à \textbf{chaque appel} · 2 ms};
    \node[cold, right=14mm of fn, yshift=-11mm] (cold)
      {\textbf{Artefacts lourds} — 253 Mo\\SDK + cache\\démarrage à froid · 8 s};

    \draw[-{Latex[length=2mm]}, gris, thick]
      (app.north east) -- node[above, font=\tiny, text=primary] {\texttt{user\_id}} (fn.north west);
    \draw[-{Latex[length=2mm]}, gris, thick]
      (fn.south west) -- node[below, font=\tiny, text=primary] {5 \texttt{article\_id}} (app.south east);
    \draw[-{Latex[length=2.4mm]}, accent2, ultra thick] (hot)  -- (fn);
    \draw[-{Latex[length=2mm]}, accent, thick, dashed]  (cold) -- (fn);
  \end{tikzpicture}

  \vspace{0.4em}
  \raggedright\scriptsize
  \begin{itemize}\itemsep2pt
    \item \pointfort{Sans état.} La Function ne conserve rien entre deux requêtes :
          le profil d'un lecteur qu'elle ne connaît pas arrive dans le paramètre
          \texttt{history} de l'appel.
    \item \pointfort{Mais pas sans mémoire.} L'instance \texttt{Recommender} est un
          \textbf{singleton} réutilisé d'une invocation à l'autre : les 253 Mo sont
          lus une seule fois par instance, au démarrage à froid, après
          \textbf{découverte dynamique} du conteneur — le chargeur énumère les
          blobs, il ne suit pas une liste figée.
    \item \pointfort{Deux accès, pas un.} Le \emph{binding} relit son blob à chaque
          invocation : ce qu'il faut pour 23 Ko qui changent toutes les heures.
          Appliqué aux 253 Mo il supprimerait le cache — \textbf{8 s par appel au
          lieu de 0,2 s}, 40 fois plus.
  \end{itemize}
\end{frame}

\begin{frame}{Ce qui sépare un service d'une démonstration}
  \framesubtitle{Quatre défaillances silencieuses, et le garde-fou de chacune}
  \centering
  \scriptsize
  \begin{tabular}{@{}p{3.3cm}p{5.1cm}p{4.2cm}@{}}
    \toprule
    \textbf{Défaillance} & \textbf{Garde-fou} & \textbf{Preuve} \\
    \midrule
    Dérive d'artefact\newline\textcolor{gris}{\tiny observée en production}
      & validité du cache jugée sur la taille \textbf{et} la date du blob — un
        modèle reconstruit de taille identique est retéléchargé
      & \texttt{test\_meme\_taille\_}\newline\texttt{mais\_blob\_plus\_recent} \\[2pt]
    Cache périmé\newline\textcolor{gris}{\tiny observée en production}
      & le conteneur est \textbf{énuméré} à chaque démarrage ; une liste figée
        avait fait manquer les artefacts ajoutés après elle
      & \texttt{list\_blobs()}, trois artefacts requis contrôlés \\[2pt]
    Catalogue incohérent
      & tout identifiant hors catalogue est filtré, au chargement \emph{et} à
        chaque requête — sinon l'indexation échoue
      & \texttt{test\_historique\_hors\_}\newline\texttt{catalogue\_filtre} \\[2pt]
    Interférence entre requêtes concurrentes
      & la surcharge \texttt{history} vit dans un \texttt{threading.local},
        remise à zéro en \texttt{finally} : le profil d'un appel ne fuit pas
        vers un autre
      & isolation par requête, \texttt{src/recommender.py} \\
    \bottomrule
  \end{tabular}

  \vspace{0.6em}
  \raggedright\scriptsize
  \pointfort{Aucune de ces quatre défaillances ne lève d'erreur.} Le service
  continue de répondre 200, avec de mauvaises recommandations — les deux
  premières ont été découvertes en comparant la réponse déployée à la réponse
  locale, pas par une exception.
\end{frame}
